// A limit of the scan, kept here so that it is stated by a test. Both constructors skip when the library is
// missing, and both run before the call, so where the call would fail the test, the test is skipped. The
// first calls a method of another file under a name of its own. The second makes the exception by
// reflection. Neither names anything the scan reads, and the scan reports nothing.
// expect: nothing
using Credentials.TestSupport;
using Xunit;

namespace Fixtures;

[Trait("Category", "Bbs")]
public sealed class Bbs2023PreparedTests
{
    public Bbs2023PreparedTests()
    {
        Hosting.Prepare();
    }

    [SkippableFact]
    public void Derives()
    {
        BbsNativeLibrary.RequireOrSkip();
    }
}

[Trait("Category", "Bbs")]
public sealed class Bbs2023ReflectedSkipTests
{
    public Bbs2023ReflectedSkipTests()
    {
        if (!BbsNativeLibrary.IsAvailable)
        {
            var quiet = typeof(SkippableFactAttribute).Assembly.GetExportedTypes().Single(type => type.IsSubclassOf(typeof(Exception)));
            throw (Exception)Activator.CreateInstance(quiet, "the native library did not load")!;
        }
    }

    [SkippableFact]
    public void Derives()
    {
        BbsNativeLibrary.RequireOrSkip();
    }
}
