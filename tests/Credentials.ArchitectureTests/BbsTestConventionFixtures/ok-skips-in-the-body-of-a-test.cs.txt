using Credentials.TestSupport;
using Xunit;

namespace Fixtures;

[Trait("Category", "Bbs")]
public sealed class Bbs2023SkipInTheBodyTests
{
    [SkippableFact]
    public void Derives_on_a_portable_build()
    {
        BbsNativeLibrary.RequireOrSkip();
        Skip.If(IsRuntimeSpecificBuild(), "a portable build only");
        Derive(Rows().Skip(1).ToArray());
    }

    [SkippableFact]
    public void Reads_the_declared_assets()
    {
        Skip.If(IsRuntimeSpecificBuild(), "a portable build only");
        var skipped = () => throw new SkipException("never thrown");
    }

    private static readonly BbsNativeLibrary.Outcome Elsewhere = BbsNativeLibrary.Outcome.Skip;

    [Theory]
    [InlineData(false, false, false, BbsNativeLibrary.Outcome.Skip)]
    public void Maps_an_outcome(bool available, bool shipped, bool onGitHubActions, BbsNativeLibrary.Outcome expected)
    {
        Assert.Equal(expected, BbsNativeLibrary.OutcomeFor(available, shipped, onGitHubActions));
    }

    [Fact]
    public void Carries_out_an_outcome()
    {
        var skip = () => BbsNativeLibrary.Enforce(BbsNativeLibrary.Outcome.Skip);
        Assert.Throws<SkipException>(skip);
    }
}
