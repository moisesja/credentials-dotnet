using Xunit;

namespace Fixtures;

public sealed class DerivedProofRoundTripTests
{
    [Fact]
    public void Through_a_helper_in_another_file()
    {
        var suite = Suites.SelectiveDisclosure();
        if (!suite.IsAvailable) return;
    }

    [Fact]
    public void Through_a_name_put_together_at_run_time()
    {
        var type = Type.GetType("DataProofsDotnet.Rdfc.DataIntegrity.Bb" + "s2023Cryptosuite, DataProofsDotnet.Rdfc", throwOnError: true)!;
        var suite = Activator.CreateInstance(type, [null])!;
        if (!(bool)type.GetProperty("IsAvailable")!.GetValue(suite)!) return;
    }
}
